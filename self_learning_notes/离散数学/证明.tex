\documentclass[12pt,a4paper]{article} % 设置文档类型为文章，字体大小为12pt，纸张大小为A4
\usepackage{ctex} % 加载ctex宏包，支持中文
\usepackage{amsmath} % 加载amsmath宏包，用于数学公式排版
\usepackage{graphicx} % 加载graphicx宏包，用于插入图片
\usepackage{geometry} % 加载geometry宏包，用于设置页面布局
\geometry{left=2.5cm,right=2.5cm,top=2.5cm,bottom=2.5cm} % 设置页面边距

\title{笔记标题} % 设置文档标题
\author{作者姓名} % 设置作者
\date{\today} % 设置日期为当前日期

\begin{document} % 文档开始
	
	\maketitle % 生成标题、作者和日期
	
	\section{引言} % 创建一级标题
	这里是引言部分的内容，用于介绍笔记的主题和背景。
	
	\section{主要内容} % 创建一级标题
	\subsection{归纳推理} % 创建二级标题
	
	
	\begin{itemize} % 创建无序列表
		\item 第一个要点
		\item 第二个要点
		\item 第三个要点
	\end{itemize}
	
	\subsection{证明方法} % 创建二级标题
	这是主要内容的第二部分，可以继续记录相关内容。
	
	\begin{enumerate} % 创建有序列表
		\item 第一步
		\item 第二步
		\item 第三步
	\end{enumerate}
	
	\section{总结} % 创建一级标题
	这里是总结部分的内容，用于对笔记的主要内容进行总结和回顾。
	
\end{document} % 文档结束
